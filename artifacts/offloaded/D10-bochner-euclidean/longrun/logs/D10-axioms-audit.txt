theorem Poincare.D10.BochnerEuclidean.bochner_identity : ∀ {n : ℕ} {u : E n → ℝ},
  ContDiff ℝ 3 u →
    ∀ (x : E n), lap (fun y => ‖grad u y‖ ^ 2) x = 2 * hessNormSq u x + 2 * inner ℝ (grad u x) (grad (lap u) x) :=
fun {n} {u} hu x =>
  have hfun := funext fun y => norm_sq_grad u y;
  Eq.mpr (id (congrArg (fun _a => lap _a x = 2 * hessNormSq u x + 2 * inner ℝ (grad u x) (grad (lap u) x)) hfun))
    (Eq.mpr
      (id
        (congrArg (fun _a => _a = 2 * hessNormSq u x + 2 * inner ℝ (grad u x) (grad (lap u) x))
          (bochner_identity_components hu x)))
      (Eq.mpr
        (id
          (congrArg (fun _a => 2 * hessNormSq u x + 2 * gradLapDot u x = 2 * hessNormSq u x + 2 * _a)
            (inner_grad_grad_lap u x)))
        (Eq.mpr
          (id (congrArg (fun _a => 2 * hessNormSq u x + 2 * _a = 2 * hessNormSq u x + 2 * _a) (gradLapDot.eq_1 u x)))
          (Eq.refl (2 * hessNormSq u x + 2 * ∑ i, D i u x * D i (lap u) x)))))
theorem Poincare.D10.BochnerEuclidean.harmonic_energy_density_subharmonic : ∀ {n : ℕ} {u : E n → ℝ},
  ContDiff ℝ 3 u →
    Harmonic u →
      ∀ (x : E n), lap (fun y => ‖grad u y‖ ^ 2) x = 2 * hessNormSq u x ∧ 0 ≤ lap (fun y => ‖grad u y‖ ^ 2) x :=
fun {n} {u} hu hh x => ⟨harmonic_lap_gradNormSq hu hh x, harmonic_lap_gradNormSq_nonneg hu hh x⟩
